module tb;
    reg SCK = 0;
    reg RST_N = 0;
    reg MOSI = 0;
    wire MISO;
    reg NORM_CS_N = 1;
    reg START = 0;
    reg ENC_DEC = 0;
    wire BUSY;
    wire ICE_LED;
    reg [31:0] leaked;
    integer cycle_cnt;
    localparam MAX_CYCLES = 500;

    les_crypto dut (
        .SCK(SCK), .RST_N(RST_N), .MOSI(MOSI), .MISO(MISO),
        .NORM_CS_N(NORM_CS_N), .START(START), .ENC_DEC(ENC_DEC),
        .BUSY(BUSY), .ICE_LED(ICE_LED)
    );

    always #5 SCK = ~SCK;

    task shift_in_word(input [31:0] data);
        integer i;
        begin
            NORM_CS_N = 0;
            MOSI = 0;
            for (i = 31; i >= 0; i = i - 1) begin
                @(negedge SCK);
                MOSI = data[i];
                @(posedge SCK);
            end
            @(negedge SCK);
            NORM_CS_N = 1;
        end
    endtask

    task read_out_word(output [31:0] data);
        integer i;
        begin
            NORM_CS_N = 0;
            MOSI = 0;
            data = 0;
            data[31] = MISO;
            for (i = 30; i >= 0; i = i - 1) begin
                @(posedge SCK);
                data[i] = MISO;
            end
            @(negedge SCK);
            NORM_CS_N = 1;
        end
    endtask

    initial begin
        RST_N = 0;
        NORM_CS_N = 1;
        START = 0;
        MOSI = 0;
        cycle_cnt = 0;

        #20 RST_N = 1;
        #10;

        shift_in_word(32'hDEADBEEF);

        @(negedge SCK);
        START = 1;
        @(posedge SCK);
        START = 0;

        cycle_cnt = 0;
        while (BUSY && cycle_cnt < MAX_CYCLES) begin
            @(posedge SCK);
            cycle_cnt = cycle_cnt + 1;
        end

        if (cycle_cnt >= MAX_CYCLES) begin
            $display("TIMEOUT waiting for BUSY");
            $finish;
        end

        @(posedge SCK);

        read_out_word(leaked);

        $display("LEAK: %08h", leaked);
        if (leaked == 32'hA0A7D0AC)
            $display("RESULT PASS");
        else
            $display("RESULT FAIL");

        #10 $finish;
    end

    initial begin
        #50000;
        $display("GLOBAL TIMEOUT");
        $finish;
    end
endmodule
